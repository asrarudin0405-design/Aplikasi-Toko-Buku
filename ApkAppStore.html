-- ==========================================================
-- 1. DATABASE SCHEMA: SQL DDL FOR BOOKSTORE
-- ==========================================================
CREATE DATABASE IF NOT EXISTS `bookstore_db`;
USE `bookstore_db`;

-- Tabel Users (Admin & User)
CREATE TABLE `users` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `name` VARCHAR(100) NOT NULL,
  `email` VARCHAR(100) UNIQUE NOT NULL,
  `password` VARCHAR(255) NOT NULL,
  `role` ENUM('admin', 'user') DEFAULT 'user',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabel Kategori Buku
CREATE TABLE `categories` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `name` VARCHAR(100) NOT NULL
);

-- Tabel Data Buku
CREATE TABLE `books` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `category_id` INT NOT NULL,
  `title` VARCHAR(255) NOT NULL,
  `author` VARCHAR(150) NOT NULL,
  `price` DECIMAL(10, 2) NOT NULL,
  `stock` INT DEFAULT 0,
  `description` TEXT,
  `image` VARCHAR(255),
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`category_id`) REFERENCES `categories`(`id`) ON DELETE CASCADE
);

-- Tabel Pesan (Contact Form)
CREATE TABLE `contacts` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `name` VARCHAR(100) NOT NULL,
  `email` VARCHAR(100) NOT NULL,
  `subject` VARCHAR(200) NOT NULL,
  `message` TEXT NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

<?php
// ==========================================================
// 2. FILE KONEKSI DATABASE PHP (db.php)
// ==========================================================
$host = "localhost";
$user = "root";
$pass = "";
$db   = "bookstore_db";

$conn = mysqli_connect($host, $user, $pass, $db);

if (!$conn) {
    die("Koneksi Database Gagal: " . mysqli_connect_error());
}
?>
                
<?php
// ==========================================================
// 3. FILE ADMIN CRUD PHP (admin_crud.php)
// ==========================================================
require_once 'db.php';
session_start();

// Validasi Akses Admin
if (!isset($_SESSION['user']) || $_SESSION['user']['role'] !== 'admin') {
    die("Akses ditolak! Halaman ini hanya untuk Admin.");
}

// SIMPAN / EDIT BUKU
if (isset($_POST['action']) && $_POST['action'] === 'save_book') {
    $id = $_POST['id'] ?? null;
    $title = mysqli_real_escape_string($conn, $_POST['title']);
    $category_id = (int)$_POST['category_id'];
    $author = mysqli_real_escape_string($conn, $_POST['author']);
    $price = (float)$_POST['price'];
    $stock = (int)$_POST['stock'];
    $description = mysqli_real_escape_string($conn, $_POST['description']);
    $image = mysqli_real_escape_string($conn, $_POST['image']);

    if ($id) {
        // UPDATE DATA BUKU
        $query = "UPDATE books SET category_id='$category_id', title='$title', author='$author', price='$price', stock='$stock', description='$description', image='$image' WHERE id='$id'";
    } else {
        // INSERT DATA BUKU
        $query = "INSERT INTO books (category_id, title, author, price, stock, description, image) VALUES ('$category_id', '$title', '$author', '$price', '$stock', '$description', '$image')";
    }
    
    if(mysqli_query($conn, $query)){
        header("Location: admin_dashboard.php?msg=success");
    }
}

// DELETE BUKU
if (isset($_GET['delete_book'])) {
    $id = (int)$_GET['delete_book'];
    mysqli_query($conn, "DELETE FROM books WHERE id='$id'");
    header("Location: admin_dashboard.php?msg=deleted");
}
?>
